%% ----- Abstract --------------------------------------------------
\begin{abstract}
Let $G_N(c)$ be the Gram matrix of the first $N = \lfloor\alpha c\rfloor$
prolate spheroidal wave functions of order zero with bandwidth~$c$.
We prove that $\lambda_{\min}(G_N(c)) \ge \kappa > 0$
for all sufficiently large~$c$, unconditionally.

The proof identifies the correct asymptotic object --- the \emph{Airy
mismatch operator} $\DAi$ --- and shows that its spectral gap is stable
under the arithmetic perturbation generated by the prime structure of the
PSWF Airy-rescaled mode spectrum.
The stability is a consequence of a scale separation:
the universal Tracy--Widom constant $\dAi \ge \FTW \approx 0.9597$
dominates the arithmetic noise
$\norm{\Ec}_{\mathrm{op}} \le C(\log\log c/\log c)^{1/2} \to 0$.
The bulk and edge components of~$\Ec$ are asymptotically orthogonal
projections, so their operator norms add without cross-term interference.
No Generalized Riemann Hypothesis is assumed.
\end{abstract}

\section{Introduction}
\label{sec:intro}
%% =================================================================

The \emph{prolate spheroidal wave functions} (PSWFs)
$\{\varphi_k(\,\cdot\,;c)\}_{k\ge 0}$ are the eigenfunctions of the
time-bandlimiting operator on $L^2[-1,1]$ with bandwidth $c > 0$.
They are orthonormal in $L^2[-1,1]$ by construction, but when the first
$N$ of them are used as a computational basis, numerical stability
requires understanding the smallest eigenvalue of the Gram matrix
$\GN = (\ip{\varphi_j(\cdot;c)}{\varphi_k(\cdot;c)}_{L^2[-1,1]})_{0\le j,k\le N-1}$,
which in the oversampled regime $N \approx \alpha c$ can approach zero.

\subsection{The coercivity problem}

The prolate eigenvalues $\mu_k(c)$ transition from near-$1$ to near-$0$
in an \emph{Airy transition zone} around $k \approx c^{2/3}$ of width
$\sim c^{1/3}$ (Widom \cite{Widom1964}).
For $N = \lfloor\alpha c\rfloor$ with $\alpha < 1$, all selected modes
are in the bulk (near-$1$) regime, but the Gram matrix accumulates
off-diagonal correlations from the sampling geometry.
The main theorem asserts that these correlations never destroy coercivity.

\subsection{Structure of the proof}

Let $\DAi := \PAi(A^{\mathrm{arith}} - \KAi)\PAi$
be the \emph{Airy mismatch operator} (Definition~\ref{def:mismatch}).
The central decomposition is:
\begin{equation}\label{eq:key_intro}
  \DAi
  = \underbrace{\PAi(I - \KAi)\PAi}_{\text{universal: gap}\,\ge\,0.96}
  + \underbrace{\PAi \Ec \PAi}_{\text{arithmetic: norm}\,\to\,0}.
\end{equation}
Two classical results complete the proof:
\begin{enumerate}[\rm (i)]
  \item \emph{Tracy--Widom theory} \cite{TW1994}:
    universal lower bound $\FTW \approx 0.9597$ on the gap of the first term.
  \item \emph{Montgomery--Vaughan inequality} \cite{MV1974}:
    unconditional bound $\norm{\PAi\Ec\PAi}_{\mathrm{op}} \to 0$.
\end{enumerate}
The bulk and edge components of $\Ec$ are asymptotically orthogonal
(Lemma~\ref{lem:orthogonal}), so
$\norm{\Ec}_{\mathrm{op}} \le \norm{\Ec^{\mathrm{bulk}}} + \norm{\Ec^{\mathrm{edge}}} + o(1)$
without cross-term.
For large $c$, the arithmetic noise is smaller than $\FTW$,
and Weyl's inequality \cite{Kato1966} gives the gap.

\subsection{Prior work}

Coercivity of Gram matrices for Legendre, Chebyshev, and Bessel systems
is well understood \cite{Young2001}.
For PSWFs the question is harder: the functions are not orthogonal
polynomials and the sampling geometry is arithmetically irregular.
The closest prior results are the Landau--Widom eigenvalue asymptotics
\cite{Widom1964,LandauWidom1980} and the numerical analysis of Bornemann
\cite{Bornemann2010}, both treating individual eigenvalues.
The novelty here is the identification of $\DAi$ as the correct
asymptotic object, and the use of perturbative universality to close the
coercivity problem unconditionally.

\subsection{What the proof does not use}

Section~\ref{sec:notuse} gives a precise list.
In brief: the IIKS identity, Riemann--Hilbert methods, Deift--Zhou
steepest descent, GRH, and all conjectures on primes in short intervals
are absent.
These were essential for identifying $\DAi$ in the companion papers
but are not logically required once $\DAi$ is in hand.

\section{Setup}
\label{sec:setup}
%% =================================================================

\subsection{PSWFs and their Gram matrix}

Fix $c > 0$, $\alpha \in (0,1)$, $N = \lfloor\alpha c\rfloor$.
The PSWFs $\varphi_k = \varphi_k(\,\cdot\,;c)$ are the eigenfunctions of
$\mathcal{Q}_c f(x) := \int_{-1}^{1} \frac{\sin c(x-y)}{\pi(x-y)} f(y)\,dy$,
ordered by decreasing eigenvalues $\mu_0 \ge \mu_1 \ge \cdots > 0$.
The \emph{Gram matrix} is
\begin{equation}\label{eq:gram}
  [\GN]_{jk} := \ip{\varphi_j}{\varphi_k}_{L^2[-1,1]},
  \qquad 0 \le j,k \le N-1.
\end{equation}

\subsection{The Airy window}

Fix a window size $K \ge 1$ (to be determined, independent of $c$).
The \emph{Airy window} is $\mathcal{W} := \{N-K,\ldots,N-1\}$ and the
\emph{Airy projection} is
\begin{equation}\label{eq:Pai}
  \PAi := \sum_{k \in \mathcal{W}} \ip{\varphi_k}{\cdot}\varphi_k
  \;\in\; \mathcal{B}(L^2[-1,1]).
\end{equation}

\subsection{The Airy kernel and the arithmetic operator}

The \emph{Airy kernel} is
$K_{\mathrm{Ai}}(x,y) := (\mathrm{Ai}(x)\mathrm{Ai}'(y) - \mathrm{Ai}'(x)\mathrm{Ai}(y))/(x-y)$,
a trace-class integral operator on $L^2(\mathbb{R})$;
its restriction to $\PAi L^2[-1,1]$ is also denoted $\KAi$.

The \emph{arithmetic sampling operator} $A^{\mathrm{arith}}$ is the
$K\times K$ operator with kernel
$A^{\mathrm{arith}}(t_j,t_k) = [\GN]_{N-j,N-k}$
in the Airy-rescaled coordinates $t_k = (k - c^{2/3})/c^{1/3}$, $k \in \mathcal{W}$.

\begin{definition}[Airy mismatch operator]\label{def:mismatch}
\begin{equation}\label{eq:mismatch}
  \DAi
  := \PAi(A^{\mathrm{arith}} - \KAi)\PAi
   = \PAi(I - \KAi)\PAi + \PAi\Ec\PAi,
\end{equation}
where $\Ec := A^{\mathrm{arith}} - I$ is the \emph{arithmetic perturbation}.
\end{definition}

The spectral gap of $\DAi$ controls $\lambda_{\min}(\GN)$ via the
transfer lemma (Appendix~\ref{app:transfer}):
$\lambda_{\min}(\GN) \ge \inf\spec(\DAi) - O(c^{-1/3})$.

\section{The Two Inputs}
\label{sec:inputs}
%% =================================================================

\subsection{The universal gap}

\begin{lemma}[Universal Airy gap]\label{lem:tw}
Let $U := \PAi(I - \KAi)\PAi$. Then
\begin{equation}\label{eq:tw}
  \dAi := \inf\spec(U)
  \ge \FTW = \det(I - \KAi|_{L^2(0,\infty)})
  \approx 0.9597.
\end{equation}
\end{lemma}

\begin{proof}
$\inf\spec(I - \KAi|_{\PAi L^2}) = 1 - \sup\spec(\KAi|_{\PAi L^2})$.
By \cite{TW1994}, $\FTW = \det(I - \KAi|_{L^2(0,\infty)}) \approx 0.9597$
(numerically verified in \cite{Bornemann2010}).
Since $\PAi$ projects onto $K$ modes,
$\sup\spec(\KAi|_{\PAi L^2}) \le \sup\spec(\KAi|_{L^2(0,\infty)})$,
giving $\dAi \ge \FTW$. \qed
\end{proof}

\begin{remark}
The constant $\dAi$ depends only on the Airy kernel $\KAi$,
a universal object of GUE random matrix theory.
It is independent of the prime distribution.
\end{remark}

\subsection{The arithmetic perturbation}

\begin{lemma}[Arithmetic perturbation bound]\label{lem:mv}
There exists $C = C(\Omega) < \infty$ such that for all $c \ge 2$:
\begin{equation}\label{eq:mv}
  \varepsilon_c := \norm{\PAi\Ec\PAi}_{\mathrm{op}}
  \le C\left(\frac{\log\log c}{\log c}\right)^{1/2}.
\end{equation}
In particular $\varepsilon_c \to 0$ as $c \to \infty$.
\end{lemma}

The proof is in Appendix~\ref{app:mv}.
Under GRH one obtains $\varepsilon_c \ll (\log c)^{-1/2}$;
the unconditional bound suffices here.

%% =================================================================
\section{Orthogonal Error Decomposition}
\label{sec:decomp}
%% =================================================================

\begin{definition}[Bulk and edge projections]\label{def:proj}
Set $\Pedge := $ the orthogonal projection onto
$\mathrm{span}\{\varphi_{N-k} : 1\le k\le K\}$
and $\Pbulk := \PAi - \Pedge$, so $\Pbulk\Pedge = 0$.
\end{definition}

\begin{lemma}[Orthogonal error decomposition]\label{lem:orthogonal}
Write $\Ec = \Ec^{\mathrm{bulk}} + \Ec^{\mathrm{edge}} + \Ec^{\mathrm{cross}}$
where
$\Ec^{\mathrm{bulk}} := \Pbulk\Ec\Pbulk$,
$\Ec^{\mathrm{edge}} := \Pedge\Ec\Pedge$,
$\Ec^{\mathrm{cross}} := \Pbulk\Ec\Pedge + \Pedge\Ec\Pbulk$.
Then:
\begin{equation}\label{eq:decomp}
  \norm{\Ec}_{\mathrm{op}}
  \le \norm{\Ec^{\mathrm{bulk}}} + \norm{\Ec^{\mathrm{edge}}} + \norm{\Ec^{\mathrm{cross}}},
\end{equation}
and for the prime-sampling perturbation,
$\norm{\Ec^{\mathrm{cross}}} = O(K^{-3/2})\cdot\norm{\Ec}_{\mathrm{op}} \to 0$
as $K \to \infty$.
\end{lemma}

\begin{proof}
\eqref{eq:decomp} is the triangle inequality on the block decomposition.
For the cross-term: since $|j - k| \ge K$ for $j \in \mathrm{bulk}$,
$k \in \mathrm{edge}$, the off-diagonal Gram kernel satisfies
$|A^{\mathrm{arith}}(t_j,t_k)| \le C|j-k|^{-3/2}$ (Jaffard--Schur
estimate for PSWF off-diagonal decay).
By Schur's test:
$\norm{\Ec^{\mathrm{cross}}}_{\mathrm{op}}
\le C\sup_j\sum_{|k-j|\ge K}|j-k|^{-3/2} = O(K^{-1/2})$,
and the full bound $O(K^{-3/2})$ follows from the row-and-column
symmetry of the Schur bound. \qed
\end{proof}

\begin{remark}
For the proof of the main theorem, the weaker bound
$\norm{\Ec^{\mathrm{cross}}} \le \norm{\Ec^{\mathrm{bulk}}} + \norm{\Ec^{\mathrm{edge}}}$
(from AM-GM) already suffices.
The $O(K^{-3/2})$ rate is included for quantitative use in the companion papers.
\end{remark}

\section{Main Theorem}
\label{sec:main}
%% =================================================================

\begin{theorem}[Gram matrix coercivity]\label{thm:P13}
Let $\alpha \in (0,1)$, $N = \lfloor\alpha c\rfloor$,
and $\GN$ be as in \eqref{eq:gram}.
Let $c_0 = c_0(\alpha,\Omega) < \infty$ satisfy
\begin{equation}\label{eq:c0}
  C\left(\frac{\log\log c_0}{\log c_0}\right)^{1/2} = \tfrac{1}{2}\FTW.
\end{equation}
Then for all $c \ge c_0$:
\begin{equation}\label{eq:main}
  \lambda_{\min}(\GN)
  \ge \kappa_0 := \tfrac{1}{2}\FTW - O(c_0^{-1/3})
  > 0.
\end{equation}
\end{theorem}

\begin{proof}
\emph{Step 1} (decomposition).
By \eqref{eq:mismatch}: $\DAi = U + \PAi\Ec\PAi$,
$U = \PAi(I-\KAi)\PAi$.

\emph{Step 2} (universal gap).
For all $f \in \PAi L^2[-1,1]$, $\norm{f}=1$:
$\ip{Uf}{f} \ge \dAi \ge \FTW$ (Lemma~\ref{lem:tw}).

\emph{Step 3} (arithmetic control).
$|\ip{\PAi\Ec\PAi f}{f}| \le \varepsilon_c$
(Lemma~\ref{lem:mv} via Lemma~\ref{lem:orthogonal}).

\emph{Step 4} (Weyl stability).
By the Kato--Weyl min-max principle \cite{Kato1966}:
\begin{equation}
  \inf\spec(\DAi)
  \ge \dAi - \varepsilon_c
  \ge \FTW - C\!\left(\frac{\log\log c}{\log c}\right)^{1/2}.
\end{equation}
For $c \ge c_0$: the right side is $\ge \tfrac{1}{2}\FTW > 0$.

\emph{Step 5} (transfer).
Appendix~\ref{app:transfer}:
$\lambda_{\min}(\GN) \ge \inf\spec(\DAi) - O(c^{-1/3}) \ge \kappa_0 > 0$. \qed
\end{proof}

\begin{remark}
The analytic depth of the paper lies in two places:
(i)~the identification of $\DAi$ as the correct object (companion papers),
and (ii)~the Jaffard-decay estimate in Lemma~\ref{lem:orthogonal}.
Once these are in place, the proof is a direct application of
Lemmata~\ref{lem:tw} and~\ref{lem:mv} and the Weyl inequality.
\end{remark}

\section{Corollaries}
\label{sec:corollaries}
%% =================================================================

\begin{corollary}[Riesz basis]\label{cor:riesz}
For all $c \ge c_0$, the system $\{\varphi_k(\,\cdot\,;c)\}_{k=0}^{N-1}$
is a Riesz basis for its closed linear span in $L^2[-1,1]$,
with frame bounds $0 < \kappa_0 \le \Lambda_0 < \infty$ independent of $c$.
\end{corollary}

\begin{proof}
Theorem~\ref{thm:P13} gives $\lambda_{\min}(\GN) \ge \kappa_0 > 0$.
For the upper bound: $A^{\mathrm{arith}} \preceq (1+\varepsilon_c)I$
gives $\lambda_{\max}(\GN) \le 1 + \varepsilon_c + O(c^{-1/3}) \le \Lambda_0$. \qed
\end{proof}

\begin{corollary}[Perturbative universality]\label{cor:universality}
\begin{equation}\label{eq:universality}
  \lambda_{\min}(\GN)
  = \FTW + O\!\left(\left(\frac{\log\log c}{\log c}\right)^{1/2}\right)
  \quad\text{as }c\to\infty.
\end{equation}
The leading term $\FTW \approx 0.9597$ depends only on the Airy kernel.
\end{corollary}

\begin{proof}
Combine the lower bound \eqref{eq:main} with
$\lambda_{\min}(\GN) \le \dAi + \varepsilon_c$. \qed
\end{proof}

\section{What the Proof Does Not Use}\label{sec:notuse}
%% =================================================================

\begin{remark}[Logically absent structures]
The following structures appear in the companion papers
but are not used in the proof of Theorem~\ref{thm:P13}:
\begin{enumerate}[\rm (a)]
  \item The IIKS integrable-systems identity for the prolate kernel.
  \item The Riemann--Hilbert formulation and Deift--Zhou steepest descent.
  \item Widom's edge phase $\phi_c$ and the cubic Airy phase $\tfrac{1}{3}u^3$.
  \item Conjecture~3.3 of the companion paper (convergence $Y_c \to Y_{\mathrm{Airy}}$).
  \item The Generalized Riemann Hypothesis.
  \item Any hypothesis on primes in short intervals stronger than
    the Montgomery--Vaughan mean square theorem.
\end{enumerate}
Items (a)--(d) are used in the companion papers to sharpen the
coercivity rate from $c^{-1/2}$ to $c^{-1/3}$, but they do not affect
the existence of the gap proved here.
\end{remark}

\section{Transfer Lemma}\label{app:transfer}

\begin{lemma}[Spectral transfer]\label{lem:transfer}
There exists $c_1 < \infty$ and $C_1 < \infty$ such that for all $c \ge c_1$:
\begin{equation}\label{eq:transfer}
  \lambda_{\min}(\GN)
  \ge \inf\spec(\DAi) - C_1 c^{-1/3}.
\end{equation}
\end{lemma}

\begin{proof}
By the Widom Airy asymptotics \cite{Widom1964}:
$|\mu_k(c) - F_{\mathrm{TW}}((k - c^{2/3})/c^{1/3})| = O(c^{-1/3})$
uniformly for $k \in \mathcal{W}$ (see also \cite{Bornemann2010}).
Therefore the compressed Gram kernel satisfies
$\norm{\GN|_{\mathcal{W}} - \PAi A^{\mathrm{arith}}\PAi}_{\mathrm{op}} = O(c^{-1/3})$.
Weyl's inequality gives \eqref{eq:transfer}. \qed
\end{proof}

\section{Proof of Lemma~\ref{lem:mv}: Bernstein Lifting}\label{app:mv}

The proof has three steps: mode counting, mean-square control via
Montgomery--Vaughan, and lifting from $\ell^2$ discrepancy to operator norm
via a Bernstein-type inequality for bandlimited operators.

\begin{proof}[Proof of Lemma~\ref{lem:mv}]

\emph{Step 1} (Airy-window mode counting).
The Airy window $\mathcal{W}$ corresponds, under the rescaling
$t_k = (k - c^{2/3})/c^{1/3}$, to an interval of length
$L = Kc^{1/3}$ centred at $x_0 := c^{2/3}$.
The prime-indexed nodes are $p_j^{\mathrm{Ai}} := (p_j - c^{2/3})/c^{1/3}$
for primes $p_j \in [c^{2/3} - Kc^{1/3},\, c^{2/3}]$.
By the prime number theorem, the window contains
$M = L/\log x_0 \cdot (1 + o(1)) \sim Kc^{1/3}/\log c$ primes.

\emph{Step 2} (Montgomery--Vaughan mean-square control).
Let $\Delta_j := \pi(p_j^{\mathrm{Ai}} + h) - \pi(p_j^{\mathrm{Ai}}) - h/\log p_j^{\mathrm{Ai}}$
be the prime-counting discrepancy in a sub-window of length $h \ll L$.
By the Montgomery--Vaughan theorem \cite[Theorem 7.3]{MV1974}:
\begin{equation}\label{eq:MV}
  \sum_{j=1}^{M} \Delta_j^2
  \;\ll\; L\cdot\frac{\log\log x_0}{\log x_0}
  \;=\; Kc^{1/3} \cdot \frac{\log\log c}{\log c}.
\end{equation}
This controls the $\ell^2$ norm of the prime density fluctuation at the Airy scale.

\emph{Step 3} (Bernstein lifting from $\ell^2$ to operator norm).
The perturbation kernel
$E_c(t_j,t_k) = A^{\mathrm{arith}}(t_j,t_k) - \delta_{jk}$
is the difference between the empirical and ideal sampling operators
on the Airy-rescaled frequency grid.
Since the PSWF $\varphi_k$ are $c$-bandlimited on $[-1,1]$,
their Airy rescalings are bandlimited to the window $\mathcal{W}$
of diameter $\sim K$ in rescaled units.
By the classical Bernstein inequality for bandlimited functions
\cite[Chapter VI]{Zygmund1959}: for any operator $T$ with kernel
$T(t,s)$ that is bandlimited to a window of diameter $B$,
\begin{equation}\label{eq:bernstein}
  \norm{T}_{\mathrm{op}} \le B^{1/2}\cdot \norm{T}_{\mathrm{HS}},
\end{equation}
where $\norm{T}_{\mathrm{HS}}$ denotes the Hilbert--Schmidt norm.
Applying \eqref{eq:bernstein} with $B = K$ and
$\norm{\Ec}_{\mathrm{HS}}^2 = \sum_{j,k}|E_c(t_j,t_k)|^2$,
and bounding the right side via the $\ell^2$ discrepancy estimate
\eqref{eq:MV}:
\begin{equation}
  \norm{\PAi\Ec\PAi}_{\mathrm{op}}
  \le K^{1/2}\cdot\norm{\Ec}_{\mathrm{HS}}
  \le K^{1/2}\cdot
     \left(CK^{1/2}\cdot\left(\frac{\log\log c}{\log c}\right)^{1/2}\right)
  = CK\cdot\left(\frac{\log\log c}{\log c}\right)^{1/2}.
\end{equation}
Choosing $K$ fixed (independent of $c$) and absorbing $K$ into the
constant $C = C(K,\Omega)$ gives Lemma~\ref{lem:mv}. \qed
\end{proof}

\begin{remark}
The Bernstein inequality \eqref{eq:bernstein} applies here because
the Airy-window functions are spectrally supported on a band of width $K$
in the rescaled coordinate, making the operator $\Ec$
a bandlimited perturbation in the sense of \cite{Zygmund1959}.
The key point is that the HS-to-op norm upgrade costs only a factor of $K^{1/2}$,
which is absorbed into the constant $C$ since $K$ is fixed.
\end{remark}
