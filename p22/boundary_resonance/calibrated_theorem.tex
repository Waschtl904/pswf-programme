\begin{theorem}[Boundary-dominated spectral structure of $G_N$]
\label{thm:main_calibrated}

Let $c > 0$, $N = \lfloor 0.38c \rfloor$.
Define the endpoint vector
$e_c = (\varphi_k(1;c))_{k\in\mathcal{I}_{\mathrm{edge}}}$
and rank-one projection
$P_{e_c} = e_c e_c^\top/\|e_c\|_2^2$.

\begin{enumerate}[\rm(i)]

\item \textup{(Rank-one boundary structure. \textbf{Rigorous.})}
  The Euler--Maclaurin formula gives
  \[
    G_N^{\mathrm{edge}} = I + F_c + R_c,
    \qquad F_c := \frac{2}{c}\,e_c e_c^\top,
  \]
  where $F_c$ is \emph{exactly} rank one.

\item \textup{(Rank-one collapse of the correction. \textbf{Measured.})}
  For $c \in \{32, 48, 64\}$: $R_c$ is ${\ge}\,95\%$ rank-one in
  direction $e_c$ (next singular value ${\le}\,0.017$), giving
  \[
    G_N^{\mathrm{edge}} = I + \lambda_{\mathrm{eff}}(c)\,P_{e_c} + R_c^\perp,
    \qquad \|R_c^\perp\|_{\mathrm{op}} \le 0.017.
  \]

\item \textup{(Spectral dominance. \textbf{Numerically supported.})}
  $\|R_c^\perp\|_{\mathrm{op}} / \lambda_{\mathrm{eff}}(c)$
  is monotonically decreasing over $c \in \{32, 48, 64\}$
  (values: $0.0064$, $0.0060$, $0.0049$),
  consistent with $\to 0$ as $c\to\infty$.

\item \textup{(Boundary localization. \textbf{Rigorous from (ii)--(iii).})}
  By the Davis--Kahan $\sin\theta$ theorem,
  \[
    \sin\angle(v_1,\,e_c/\|e_c\|)
    \le
    \frac{\|R_c^\perp\|_{\mathrm{op}}}{\lambda_{\mathrm{eff}}(c) - \alpha_c}
    \le 0.0095.
  \]
  Numerically: $|\langle v_1, e_c/\|e_c\|\rangle| \ge 0.9975$
  for all $c \in \{32,48,64\}$.

\item \textup{(Bulk stability. \textbf{Standard EM accuracy.})}
  $\|G_N^{\mathrm{bulk}} - I\|_{\mathrm{op}} = O(c^{-1/2})$.

\end{enumerate}
\end{theorem}

\begin{remark}[Honest status of each part]
\label{rem:honest_status}
Parts (i), (iv), (v) are rigorous.
Part (ii) is numerically established for $c \le 64$;
its analytic proof requires uniform WKB endpoint asymptotics.
Part (iii) is supported by WKB blow-up and numerical trend,
but the precise asymptotic rate $c^{1.45}$ is not analytically derived.
The logical weight rests on (i) and (iv):
given the numerical Dominance (iii), localization (iv) is fully rigorous.
\end{remark}
