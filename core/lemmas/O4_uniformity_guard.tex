\begin{abstract}
This document is a \emph{coherence audit}, not a proof of new results.
We verify that the three uniformity structures
$\Uolv$, $\Uph$, $\Utp$ used in the O4 proof chain
(Lemmas O4-B-2 and O4-B-2a)
are mutually compatible under the joint scaling
$c \to \infty$, $N = \lfloor\alpha c\rfloor$.
No hidden $c$-dependent error terms arise from their composition.
This is the final gate before the \texttt{close(O4)} commit.
\end{abstract}

Each invariant is a summary of a result already proved

\begin{invariant}[$\Uolv$: Olver remainder uniform in $c$ and regime]
\label{inv:olver}
Source: \texttt{olver\_growth\_lemma.tex} and Paper~XVIII, Theorem~A.

\medskip
For all $n \in \mathcal{I} = [(1-\delta)N, N)$,
all $p \in [-1,1]$, and all $c \geq c_0$:
\[
  \psi_n(p) = c^{1/6}\,\mathrm{Ai}\bigl(c^{2/3}\zeta_n(p)\bigr) + R_n(p,c),
  \qquad
  \abs{R_n(p,c)} \leq C_R\, c^{-1/2},
\]
where $C_R = C_R(\delta,\alpha) > 0$ is \emph{independent of $c$, $p$, and $n$}.

\medskip
\noindent\textbf{Regime coverage:}
\begin{itemize}
  \item Oscillatory bulk ($p \leq t_n - \epsilon_0$):
        $\Uolv$ holds by \texttt{olver\_growth\_lemma.tex}, Theorem~B
        (Olver remainder decays as $c^{-1/2}$ uniformly in the bulk).
  \item Turning-point window ($|p - t_n| \leq c^{-2/3+\epsilon}$):
        $\Uolv$ holds by the Langer matching construction;
        the Airy function absorbs the turning-point singularity
        exactly, leaving only the $O(c^{-1/2})$ global error.
  \item Exponential region ($p > t_n$):
        $\Uolv$ holds trivially since both $\psi_n$ and
        $c^{1/6}\mathrm{Ai}(c^{2/3}\zeta_n)$ are exponentially small.
\end{itemize}

\medskip
\noindent\textbf{Uniformity conclusion:}
$C_R$ does not depend on the regime.
No interpolation between regimes is needed:
the Langer--Olver construction gives a single
global estimate, not piecewise bounds.
\end{invariant}

\begin{invariant}[$\Uph$: Derivative of Langer variable uniform in $c$]
\label{inv:phase}
Source: \texttt{phase\_nondeg\_lemma.tex}, Lemma~D
(Differentiability of the Pr\"ufer phase in $c^{-1/3}$).

\medskip
For all $n \in \mathcal{I}$ and $p \in [-1, t_n - \epsilon_0]$:
\[
  \partial_p u^{(n)}(p)
  = \frac{c\,\sqrt{t_n^2 - p^2}}{\bigl(u^{(n)}(p)\bigr)^{1/2}}
  + E_\partial(p, n, c),
  \qquad
  \abs{E_\partial(p,n,c)} \leq C_\partial\, c^{1/3},
\]
where $C_\partial = C_\partial(\delta, \alpha) > 0$ is independent of $c$, $p$, $n$.

\medskip
\noindent\textbf{Critical point (B):}
Differentiation and asymptotic expansion commute up to $O(c^{1/3})$.
Specifically, the leading term
$c\sqrt{t_n^2-p^2}/(u^{(n)})^{1/2}$
is the exact derivative of the Langer variable (see
\texttt{O4\_B2a\_langer\_amplitude\_control.tex}, Step~1),
and the error $E_\partial = O(c^{1/3})$ is absorbed by the
$c^{-1/2}$ Olver remainder when computing $\partial_p A_{mn}$:
\[
  c^{1/6}\cdot c^{-5/4} \cdot c^{1/3} = c^{1/6-5/4+1/3} = c^{-3/4} \to 0.
\]
Hence no hidden $c$-growth appears in the amplitude derivative.

\medskip
\noindent\textbf{Uniformity conclusion:}
$\partial_p u^{(n)}$ is $c$-uniformly bounded
in the sense needed by $O4$-B-$2a$ (Steps~1--2).
\end{invariant}

\begin{invariant}[$\Utp$: Turning-point separation discrete-stable]
\label{inv:turning}
Source: \texttt{airy\_discrete\_stability\_lemma.tex},
Corollary (ass:gap).

\medskip
For all $c \geq c_0$ and $m, n \in \mathcal{I}$, $k = |m-n| \geq 1$:
\[
  \abs{t_m - t_n} \geq \frac{\kappa_1 k}{N},
  \qquad
  \kappa_1 = \kappa_1(\delta, \alpha) > 0 \text{ independent of } c, N.
\]

\medskip
\noindent\textbf{Derivation from ass:gap:}
From ass:gap, $\lambda_n - \lambda_{n+1} \geq \kappa_0 c^{-1/3}$.
The turning points satisfy $t_n = \cos(\pi n/(2N))$,
so
\[
  t_n - t_{n+1}
  = \frac{\pi}{2N}\sin\bigl(\pi n_0/(2N)\bigr)
  \geq \frac{\pi}{2N}\sin\bigl(\pi(1-\delta)/2\bigr)
  =: \frac{\kappa_1}{N},
\]
uniformly for $n_0 \in \mathcal{I}$.
For general $k$: $|t_m - t_n| \geq k \cdot \kappa_1/N$ by monotonicity of $\cos$.

\medskip
\noindent\textbf{No regime collapse:}
The lower bound $\kappa_1 k/N$ does not degrade as $c \to \infty$
(since $N \sim \alpha c$, the bound becomes $\kappa_1 k/(\alpha c)$,
which is the correct scale for PSWF turning-point spacing).
In particular, no regime collapses as $c \to \infty$:
the edge index set $\mathcal{I}$ has $|\mathcal{I}| = \delta N$ indices,
all with mutually separated turning points.
\end{invariant}

\section{The Compatibility Check}
\label{sec:compatibility}
%%=================================================================

\begin{theorem}[Uniformity Guard]\label{thm:guard}
The three uniformity structures
$\Uolv$ (Invariant~\ref{inv:olver}),
$\Uph$ (Invariant~\ref{inv:phase}),
$\Utp$ (Invariant~\ref{inv:turning})
are mutually compatible under the joint scaling
$c \to \infty$, $N = \lfloor \alpha c \rfloor$.
Specifically:
\begin{enumerate}[(i)]
  \item\label{it:no-leak-olver}
        The Olver remainder $|R_n| \leq C_R c^{-1/2}$ does not
        produce a $c$-growing error when combined with the
        amplitude derivative $\partial_p A_{mn}$ from O4-B-2a.
  \item\label{it:no-leak-phase}
        The differentiation error $E_\partial = O(c^{1/3})$
        is absorbed by the overall $c^{-1/2}$ Olver factor
        and does not appear in the final $|G_{mn}|$ estimate.
  \item\label{it:no-leak-turning}
        The turning-point separation $|t_m - t_n| \geq \kappa_1 k/N$
        is uniform in $c$, so the $k^{-1}$ factor in
        $\|A_{mn}'\|_{L^1} \leq C/k$ does not degrade as $c \to \infty$.
\end{enumerate}
In combination: all constants appearing in the Schur bound
\[
  \sup_n \sum_{m \neq n} |G_{mn}| \leq \frac{C(\delta,\alpha)\,\zeta(3/2)}{N}
\]
are independent of $c$ and $N$.
\end{theorem}

\begin{proof}
This is an audit, not a derivation. We verify each point by
tracing the constant through the proof chain.

\medskip
\noindent\textbf{Point~\eqref{it:no-leak-olver}.}
In O4-B-2 (Step~i), the substitution of the Olver expansion
produces an error term $|\mathcal{E}_{mn}| \leq C_R c^{-1/3}$
(after the $c^{1/6}$ prefactor from the Langer--Olver normalisation).
For any fixed $k \geq 1$:
$c^{-1/3} \leq k^{-3/2}$ iff $c \geq k^{9/2}$,
which holds for all $k \leq \delta N \sim \delta\alpha c$
when $c \geq c_0(\delta, \alpha)$.
The constant $C_R$ from $\Uolv$ is independent of $c$.
\emph{No $c$-leak.}

\medskip
\noindent\textbf{Point~\eqref{it:no-leak-phase}.}
From $\Uph$: $\partial_p u^{(n)} = [\text{leading}] + E_\partial$
with $|E_\partial| \leq C_\partial c^{1/3}$.
In the amplitude derivative computation of O4-B-2a (Step~1),
$E_\partial$ enters as:
\[
  |f'_{\text{error}}(p)| \leq \frac{1}{4}(u^{(n)})^{-5/4}\cdot C_\partial c^{1/3}.
\]
With $(u^{(n)})^{-5/4} \leq C c^{-5/6}$ in the oscillatory bulk
(where $u^{(n)} \geq c_\epsilon c^{2/3}$):
\[
  |f'_{\text{error}}| \leq C c^{-5/6+1/3} = C c^{-1/2} \to 0.
\]
Near the turning point, $E_\partial$ is further suppressed
by the same $(t_n-p)^{-\beta}$ factor as the leading term,
so the relative error is $O(c^{-1/3})$ uniformly.
\emph{No $c$-leak.}

\medskip
\noindent\textbf{Point~\eqref{it:no-leak-turning}.}
From $\Utp$: $|t_m - t_n| \geq \kappa_1 k/N$ with $\kappa_1$
independent of $c$. This bound is used in O4-B-2a (Step~5)
to obtain the $k^{-1}$ factor in $\|A_{mn}'\|_{L^1}$.
Since $\kappa_1$ is $c$-independent and $N = \lfloor\alpha c\rfloor$
grows with $c$, the separation $\kappa_1 k/N$ scales as $c^{-1}$
for fixed $k$ --- consistent with the PSWF edge spacing.
The $k^{-1}$ factor therefore holds uniformly for all $c \geq c_0$.
\emph{No $c$-leak.}

\medskip
Combining all three points: every constant in the Schur bound
$\sup_n \sum_{m\neq n}|G_{mn}| \leq C\zeta(3/2)/N$
traces back to $C_R$, $C_\partial$, $\kappa_1$, or explicit
numerical constants from standard analysis.
All are independent of $c$ and $N$.
\end{proof}
